% Slide 4: Motivation (Refactored from Introduction)
    \begin{frame}{Motivation: Resist Catastrophic Forgetting
}
        % Fine-tuning Guided by Mechanistic Insights
        % \framesubtitle{From indiscriminate updates to mechanistic-guided implanting}
        \begin{columns}[T]
            \begin{column}{0.48\textwidth}
                \begin{alertblock}{Problem: Indiscriminate Updates}
                    Conventional fine-tuning methods (e.g., LoRA) is \textbf{oblivious} to crucial weights:
                    \begin{itemize}
                        \item Updates are performed blindly across modules.
                        \item Optimization perturbs \alert{critical memory traces}.
                        \item Result: Degraded generalization.
                    \end{itemize}
                \end{alertblock}
            \end{column}
            
            \begin{column}{0.48\textwidth}
                \begin{exampleblock}{The Opportunity: Knowledge Localization \& Redundancy}
                    Mechanistic studies reveal:
                    \begin{itemize}
                        \item Internal knowledge is both \textbf{identifiable} and \textbf{editable}.
                        \item LLMs exhibit significant \textbf{parameter redundancy}.
                    \end{itemize}
                \end{exampleblock}
            \end{column}
        \end{columns}

        
    \end{frame}
    
    \begin{frame}{Next Step of PEFT: From Indiscriminate Updates to Mechanistic-Guided Adaptation}
        \textbf{Our Hypothesis:} 
        By identifying \alert{low-contributing weights}, we can repurpose them as capacity for new knowledge with \textbf{minimal damage} to existing competencies.
    \end{frame}
    % Slide 5: LoKI Framework
    \section{LoKI Framework}
    \begin{frame}{Major Advantages of LoKI}
        
        % \textbf{LoKI offers several major advantages:}
        
        % \vspace{0.3cm}
        
        \begin{enumerate}
            \item \textbf{Superior Balance Between CF and Task-Specific Performance}
            \begin{itemize}
                \item Fine-tuning LLMs of two representative sizes with LoKI effectively mitigates CF, outperforming state-of-the-art PEFT methods while maintaining strong task performance.
            \end{itemize}
            
            \vspace{0.2cm}
            
            \item \textbf{Intrinsically Parameter-Efficient}
            \begin{itemize}
                \item LoKI updates only a subset of the model's original parameters and allows for explicit control over the number of trainable weights.
            \end{itemize}
            
            \vspace{0.2cm}
            
            \item \textbf{Synergistic with Other Tuning Methods}
            \begin{itemize}
                \item In addition to adapting directly for downstream tasks, LoKI can be combined with existing parameter-efficient tuning techniques such as Low-Rank Adaptation (LoRA), further reducing the number of trainable parameters.
            \end{itemize}
        \end{enumerate}
    \end{frame}
    
    % Slide 6: LoKI Framework Details
    \begin{frame}{LoKI: Low-damage Knowledge Implanting}
        \framesubtitle{A staged fine-tuning process}
        
        \begin{columns}
            \begin{column}{0.5\textwidth}
                \begin{center}
                    \includegraphics[width=\textwidth]{assets/pipeline.pdf}
                    \\ \scriptsize \textbf{Figure 1:} Schematic illustration of the staged fine-tuning process in LoKI.

                \end{center}
            \end{column}
            
            \begin{column}{0.5\textwidth}
                % \textbf{Three Stages to Mitigate CF:}
                
                % \vspace{0.3cm}
                
                \begin{description}
                    \item[1. Analysing] Use \textbf{KVA (Knowledge Vector Attribution)} to evaluate each node's contribution to pre-trained behavior.
                    \item[2. Selecting] Apply \textbf{Layer-Balanced Strategy} to choose trainable nodes while preserving hierarchical stability.
                    \item[3. Implanting] Partition weights: Freeze critical ones ($W_{\setminus S}$), update only low-damage ones ($W_S$).
                \end{description}
            \end{column}
        \end{columns}

    \end{frame}

% Slide: Stage 1 - Knowledge Vector Attribution (KVA)
    \begin{frame}{Stage 1: Knowledge Vector Attribution (KVA)}
        % \framesubtitle{Mechanistic interpretation of FFN as Memory}
        
        \begin{columns}[T]
            % --- Left Column: FFN as KV Pair ---
            \begin{column}{0.48\textwidth}
                \textbf{Mechanistic Interpretation of FFN as Key-Value Memory}
                \begin{itemize}
                    \item \(\text{FFN}(x) = W_{down} \sigma(W_{up} x)\)

                    \item \textbf{Keys ($W_{up}$):} Detect input patterns or features.
                    \item \textbf{Values ($W_{down}$):} Store interpretable concepts.
                \end{itemize}

                \begin{block}{Knowledge Vector}
                    We treat the $j$-th column of $W_{down}$ as a \alert{Knowledge Vector} ($\mathbf{v}_j$).
                    
                    Each vector $\mathbf{v}_j$ acts as an atomic unit of stored knowledge.
                \end{block}
            \end{column}

            % --- Right Column: KVA Method ---
            \begin{column}{0.48\textwidth}
                \textbf{The KVA Method}
                \begin{itemize}
                    \item Based on \alert{Integrated Gradients (IG)} to quantify the contribution of individual knowledge vectors to the model's \textbf{pre-trained behavior} (general tasks).
                    \item Path-integrated attribution of node $j$ at layer $l$:
                \end{itemize}

                \begin{equation*}
                    \text{Attr}_{l,j}(\mathbf{x}) = \int_{\alpha=0}^{1} \frac{\partial \mathcal{L}(\alpha \cdot z_{l,j})}{\partial z_{l,j}} d\alpha
                \end{equation*}
                
                % \begin{itemize}
                %     \item $\mathcal{L}$: Model output logit.
                %     \item $z_{l,j}$: Activation of the $j$-th node.
                % \end{itemize}

            \end{column}
        \end{columns}
    \end{frame}

                    \item \alert{Clustering Effect:} Both high and low-contribution nodes are \textbf{densely concentrated in the same layers}.
                    \item \alert{Structural Bias:} Distribution is not random; modifying distant layers can "override" information flow downstream.
                \end{itemize}

    % --- Stage 2: Slide 2 (Layer-Balanced Strategy Details) ---
    \begin{frame}{Stage 2: Layer-Balanced Strategy}
        % \framesubtitle{How to choose trainable parameters}
        
        \textbf{Objective:} Allocate an equal quota of trainable parameters to each layer to preserve hierarchical stability.
        
        \begin{block}{Implementation Steps}
            \begin{enumerate}
                \item \textbf{Quota Allocation:} Define total trainable slots $T$ (hyperparameter $q\%$) and set $k_l = T/L$ per layer.
                \item \textbf{Per-Sample Selection:} For each sample $t$, select $k_l$ nodes with the \alert{lowest} KVA values.
                \item \textbf{Frequency Aggregation:} Tally selection frequencies $c_{l,i}$ across all $N$ samples.
                \item \textbf{Final Selection:} Choose nodes with the highest frequencies in each layer to form the final set $S$.
            \end{enumerate}
        \end{block}
        
        \begin{alertblock}{Why it works}
            Ensures that newly implanted knowledge \textbf{conforms to the hierarchical relationship} and avoids disproportionately disturbing critical layers.
        \end{alertblock}
    \end{frame}
    
    % Slide 7: Stage 3: Implanting
    \begin{frame}{Stage 3: Implanting}
        % \framesubtitle{Parameter partitioning}
        
        For each layer $l$, decompose $W_{down}^{(l)}$:
        \begin{equation*}
            W_{down}^{(l)} = [W_{S_l}; W_{\setminus S_l}] \rightarrow \begin{cases} W_{S_l} & \text{\alert{Active (Low-contribution)}} \\ W_{\setminus S_l} & \text{Frozen (High-contribution)} \end{cases}
        \end{equation*}

        \begin{block}{Implementation Options}
            \begin{enumerate}
                \item \textbf{LoKI Linear:} Directly update the selected original weights.
                \item \textbf{LoKI-LoRA:} Parameterize the updates as low-rank matrices:
                \begin{equation*}
                    \Delta W_{S_l} = A_l B_l
                \end{equation*}
            \end{enumerate}
        \end{block}
        \alert{Result:} Transforms LoKI into a module-wise process, easy to integrate into existing pipelines (e.g., Llama-Factory).
    \end{frame}
