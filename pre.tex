    \section{LoKI 框架}
    \begin{frame}{LoKI 的主要优势}
        
        % \textbf{LoKI offers several major advantages:}
        
        % \vspace{0.3cm}
        
        \begin{enumerate}
            \item \textbf{在灾难性遗忘与任务性能之间取得更优平衡}
            \begin{itemize}
                \item 在两种代表性规模的大语言模型上进行 LoKI 微调，可以有效缓解灾难性遗忘；在保持较强任务性能的同时，性能优于当前先进的参数高效微调方法。
            \end{itemize}
            
            \vspace{0.2cm}
            
            \item \textbf{天然具备参数高效性}
            \begin{itemize}
                \item LoKI 只更新模型原始参数的一个子集，并允许显式控制可训练权重的数量。
            \end{itemize}
            
            \vspace{0.2cm}
            
            \item \textbf{可与其他调优方法协同工作}
            \begin{itemize}
                \item 除了直接适配下游任务外，LoKI 还可以与低秩适配（LoRA）等现有参数高效调优技术结合，进一步减少可训练参数的数量。
            \end{itemize}
        \end{enumerate}
    \end{frame}

% Slide: Stage 1 - 知识向量 Attribution (KVA)
    \begin{frame}{阶段 1：知识向量归因（KVA）}
        % \framesubtitle{Mechanistic interpretation of FFN as Memory}
        
        \begin{columns}[T]
            % --- Left Column: FFN as KV Pair ---
            \begin{column}{0.48\textwidth}
                \textbf{将 FFN 解释为键值记忆的机制视角} \cite{geva2021}
                \begin{itemize}
                    \item \(\text{FFN}(x) = W_{down} \sigma(W_{up} x)\)

                    \item \textbf{键（$W_{up}$）：}检测输入模式或特征。
                    \item \textbf{值（$W_{down}$）：}存储可解释概念。
                \end{itemize}

                \begin{block}{知识向量}
                    我们将 $W_{down}$ 的第 $j$ 列视为一个\alert{知识向量}（$\mathbf{v}_j$）。
                    
                    每个向量 $\mathbf{v}_j$ 都是一个独立的知识存储单元。
                \end{block}
            \end{column}

            % --- Right Column: KVA Method ---
            \begin{column}{0.48\textwidth}
                \textbf{KVA 方法}
                \begin{itemize}
                    \item 基于\alert{积分梯度（IG）} \cite{sundararajan2017}，量化各知识向量对模型\textbf{预训练行为}（通用任务）的贡献。
                    \item 第 $l$ 层节点 $j$ 的路径积分归因：
                \end{itemize}

                \begin{equation*}
                    \text{Attr}_{l,j}(\mathbf{x}) = \int_{\alpha=0}^{1} \frac{\partial \mathcal{L}(\alpha \cdot z_{l,j})}{\partial z_{l,j}} d\alpha
                \end{equation*}


            \end{column}
        \end{columns}
    \end{frame}

    % --- Stage 2: Slide 2 (Layer-Balanced Strategy Details) ---
    \begin{frame}{阶段 2：分层平衡策略}
        % \framesubtitle{How to choose trainable parameters}
        
        \textbf{目标：}为每层分配相同的可训练参数配额，以保持层级结构稳定性 \cite{sun2025a}.
        
        \begin{block}{实施步骤}
            \begin{enumerate}
                \item \textbf{配额分配：}定义可训练位置总数 $T$（超参数 $q\%$），并为每层设置 $k_l = T/L$。
                \item \textbf{逐样本选择：}对于每个样本 $t$，选择 KVA 值\alert{最低}的 $k_l$ 个节点。
                \item \textbf{频次聚合：}统计所有 $N$ 个样本中的选择频次 $c_{l,i}$。
                \item \textbf{最终选择：}在每层选择频次最高的节点，组成最终集合 $S$。
            \end{enumerate}
        \end{block}
        
        \begin{alertblock}{有效原因}
            确保新植入的知识\textbf{符合层级关系}，避免过度扰动关键层。
        \end{alertblock}
    \end{frame}
    
    % Slide 7: 阶段 3：知识植入
    \begin{frame}{阶段 3：知识植入}
        % \framesubtitle{Parameter partitioning}
        
        对于每一层 $l$，将 $W_{down}^{(l)}$ 分解为：
        \begin{equation*}
            W_{down}^{(l)} = [W_{S_l}; W_{\setminus S_l}] \rightarrow \begin{cases} W_{S_l} & \text{\alert{激活（低贡献）}} \\ W_{\setminus S_l} & \text{冻结（高贡献）} \end{cases}
        \end{equation*}

        \begin{block}{实现方式}
            \begin{enumerate}
                \item 直接更新选定的原始权重。
                \item \textbf{LoKI-LoRA:} 将更新参数化为低秩矩阵 \cite{hu2022lora}:
                \begin{equation*}
                    \Delta W_{S_l} = A_l B_l
                \end{equation*}
            \end{enumerate}
        \end{block}
        \alert{结果：}LoKI 被转化为按模块执行的流程，易于集成到现有流水线（例如 Llama-Factory）中。
    \end{frame}
